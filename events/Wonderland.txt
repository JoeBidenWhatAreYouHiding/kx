country_event = { # America is reunited
	id = wlandevent.20
	immediate = {
		log = "[GetDateText]: [Root.GetName]: event wlandevent.20"
	}
	title = wlandevent.20.t
	desc = wlandevent.20.d
	picture = GFX_report_event_USA_congress

	fire_only_once = yes

	trigger = {
		AND = {
			tag = CGL
			has_country_flag = is_wonderland
		}
		owns_state = 361 #Maryland
		owns_state = 371 #Louisiana
		owns_state = 395 #Illinois
		OR = {
			owns_state = 797 #San Francisco
			AND = {
				CAL = { is_in_faction = yes }
				has_global_flag = CAL_seperatist_claim
			}
		}
		owns_state = 932 #Massachusetts
		NOT = {
			country_exists = TEX
			country_exists = CSA
			country_exists = USA
			country_exists = CAL
			country_exists = COR
			country_exists = WCC
			AND = {
				NEE = { 
					exists = yes
					has_country_flag = NEE_claimed_america
					is_ally_of_ROOT = no
				}
			}
			AND = {
				NYC = {
					exists = yes
					has_country_flag = NYC_can_invade_america
					is_ally_of_ROOT = no
				}
			}
			AND = {
				BBR = {
					exists = yes
					has_country_flag = BBR_claimed_america
					is_ally_of_ROOT = no
				}
			}
			AND = {
				HAW = {
					exists = yes
					has_country_flag = HAW_claimed_america
					is_ally_of_ROOT = no
				}
			}
		}
		controls_all_owned_states = yes
	}

	option = { # We are reunited (nice route)
		name = wlandevent.20.a
		trigger = {
			has_completed_focus = WLD_WR1
		}
		set_country_flag = WLD_UNITED_AMERICA
		#set_cosmetic_tag = wonderland whatever but nice
		if = {
			limit = {
				CAL = {
					is_in_faction = yes
					exists = yes
				}
				has_global_flag = CAL_seperatist_claim
			}
			CAL = {
				every_owned_state = {
					FROM.OWNER = { 
						add_state_core = ROOT
					}
				}
			}
		}
		every_state = {
			limit = {
				is_united_states = yes
			}
			add_core_of = ROOT
		}
	}
	
	option = { # We are reunited (mean route)
		name = wlandevent.20.b
		trigger = {
			has_completed_focus = WLD_WR2
		}
		set_country_flag = WLD_UNITED_AMERICA
		#set_cosmetic_tag = wonderland whatever but mean
		if = {
			limit = {
				CAL = {
					is_in_faction = yes
					exists = yes
				}
				has_global_flag = CAL_seperatist_claim
			}
			CAL = {
				every_owned_state = {
					FROM.OWNER = { 
						add_state_core = ROOT
					}
				}
			}
		}
		every_state = {
			limit = {
				is_united_states = yes
			}
			add_core_of = ROOT
		}
	}
}
