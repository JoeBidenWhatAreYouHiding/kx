country_event = { #Leaving Mitteleuropa
	id = germit.4
	immediate = { log = "[GetDateText]: [Root.GetName]: event germit.4" }
	title = germit.4.t

	desc = {
		text = germit.4.d.GER_is_dead
		trigger = {
			NOT = { country_exists = GER }
		}
	}
	desc = {
		text = germit.4.d.socialist
		trigger = {
			country_exists = GER
			AND = {
				has_socialist_government = yes
				NOT = { tag = SPR }
			}
		}
	}
	desc = {
		text = germit.4.d.war_with_GER
		trigger = {
			has_socialist_government = no
			has_war_with = GER
		}
	}
	desc = {
		text = germit.4.d.allied_with_rival
		trigger = {
			has_socialist_government = no
			NOT = { has_war_with = GER }
			AND = {
				is_in_faction = yes
				NOT = { is_in_faction_with = GER }
			}
		}
	}
	desc = {
		text = germit.4.d.left_mitteleuropa
		trigger = {
			has_socialist_government = no
			NOT = {
				has_war_with = GER
				has_country_flag = GER_kicked_from_mitteleuropa
			}
			is_in_faction = no
		}
	}
	desc = {
		text = germit.4.d.kicked_out
		trigger = {
			has_country_flag = GER_kicked_from_mitteleuropa
		}
	}

	picture = GFX_report_event_generic_read_write

	is_triggered_only = yes

	trigger = {
		GER_is_in_mitteleuropa = yes
	}

	option = {
		name = germit.4.a
		GER_remove_from_mitteleuropa = yes
		if = {
			limit = {
				NOT = {
					has_war_with = GER
					has_country_flag = GER_kicked_from_mitteleuropa
				}
			}
			GER = { country_event = germit.5 }
		}
		clr_country_flag = GER_kicked_from_mitteleuropa
	}
}

country_event = { #FROM Leaves Mitteleuropa!
	id = germit.5
	immediate = { log = "[GetDateText]: [Root.GetName]: event germit.5" }
	title = germit.5.t

	desc = {
		text = germit.5.d.socialist
		trigger = {
			FROM = { has_socialist_government = yes }
		}
	}
	desc = {
		text = germit.5.d.allied_with_rival
		trigger = {
			FROM = {
				has_socialist_government = no
				AND = {
					is_in_faction = yes
					NOT = { is_in_faction_with = GER }
				}
			}
		}
	}
	desc = {
		text = germit.5.d.left_mitteleuropa
		trigger = {
			FROM = {
				has_socialist_government = no
				is_in_faction = no
			}
		}
	}

	picture = GFX_report_event_generic_read_write

	is_triggered_only = yes

	option = { # Yes
		name = germit.5.a
		if = {
			limit = {
				FROM = { has_socialist_government = yes }
			}
			add_opinion_modifier = {
				target = FROM
				modifier = KR_outraged
			}
		}
		else = {
			add_opinion_modifier = {
				target = FROM
				modifier = KR_hates_govt
			}
		}
	}
}

country_event = { #Join Mitteleuropa?
	id = germit.6
	immediate = { log = "[GetDateText]: [Root.GetName]: event germit.6" }
	title = germit.6.t

	desc = {
		text = germit.6.d.independent
		trigger = { is_subject = no }
	}
	desc = {
		text = germit.6.d.subject
		trigger = { is_subject = yes }
	}

	picture = GFX_report_event_generic_read_write

	trigger = {
		is_in_faction_with = GER
		GER_is_in_mitteleuropa = no
		NOT = {
			has_country_flag = GER_kicked_from_mitteleuropa
			has_country_flag = UKR_has_left_mitteleuropa
			any_enemy_country = { is_ally_with = GER }
		}
	}

	is_triggered_only = yes

	option = { # Yes
		name = germit.6.a
		ai_chance = {
			factor = 80
			modifier = {
				factor = 0
				has_game_rule = {
					rule = GER_mitteleuropa
					option = GER_MITTELEUROPA_NOT_STONKS
				}
			}
			modifier = {
				factor = 0
				tag = ROM
				is_subject = no
			}
		}
		add_political_power = -25
		GER_add_to_mitteleuropa = yes
		if = {
			limit = { is_subject = no }
			GER = {
				country_event = { id = germit.7 days = 1 }
			}
		}
		hidden_effect = {
			every_subject_country = { country_event = germit.6 }
		}
	}

	option = { # No
		name = germit.6.b
		trigger = { is_subject = no }
		ai_chance = {
			factor = 20
			modifier = {
				factor = 0
				ic_ratio = { tag = GER ratio < 0.1 }
			}
			modifier = {
				factor = 0
				tag = HOL
			}
			modifier = {
				factor = 0
				has_game_rule = {
					rule = GER_mitteleuropa
					option = GER_MITTELEUROPA_STONKS
				}
			}
		}
		add_political_power = 25
		GER = {
			country_event = { id = germit.8 days = 1 }
		}
	}
}

		name = germit.22.a

		hidden_effect = {
			GER_decrease_mitteleuropa_buff = yes

			GER_recalculate_mitteleuropa_effects = yes
		}
	}

			GER_increase_mitteleuropa_buff = yes
			GER_recalculate_mitteleuropa_effects = yes
		}
		add_political_power = -70
	}
}
