	}
	divide_temp_variable = { temp_mp_ser = temp_mp_aus }
	clamp_temp_variable = {
		var = temp_mp_ser
		min = 0
		max = 1
	}
	set_temp_variable = { temp_mp_required = 0.5 }
	if = {
		limit = { AUS = { AUS_has_any_bad_military_idea = yes } }
		add_to_temp_variable = { temp_mp_required = -0.1 }
	}
	if = {
		limit = { date > 1939.01.01 }
		add_to_temp_variable = { temp_mp_required = -0.25 }
	}
	#log = "SER_ROM_destablize_austria_trigger: temp_mp_ser: [?temp_mp_ser]"
	OR = {
		fifth_balkan_war = yes
		AND = {
			check_variable = { temp_mp_ser > temp_mp_required }
			NOT = {
				any_enemy_country = { limit = { is_major = yes } }
			}
		}
	}
	NOT = {
		has_global_flag = fifth_balkan_war_ended
		has_global_flag = AUS_collapsed
	}
