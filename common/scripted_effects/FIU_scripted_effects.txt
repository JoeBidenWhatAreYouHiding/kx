FIU_add_risk_effect = {
	custom_override_tooltip = {
		tooltip = FIU_add_risk_effect_tt
		add_to_variable = { FIU_risk = temp_FIU_risk }
		clamp_variable = {
			var = FIU_risk
			min = 0
			max = 1
		}
		if = {
			limit = { check_variable = { FIU_risk > 0.8 } }
			set_variable = { FIU.headsup_display_var = 5 }
		}
		else_if = {
			limit = { check_variable = { FIU_risk > 0.6 } }
			set_variable = { FIU.headsup_display_var = 4 }
		}
		else_if = {
			limit = { check_variable = { FIU_risk > 0.4 } }
			set_variable = { FIU.headsup_display_var = 3 }
		}
		else_if = {
			limit = { check_variable = { FIU_risk > 0.2 } }
			set_variable = { FIU.headsup_display_var = 2 }
		}
		else = {
			set_variable = { FIU.headsup_display_var = 1 }
		}
	}
}

FIU_add_one_successful_plunder = {
	custom_effect_tooltip = FIU_raid_succeeds
	add_to_variable = { FIU_successful_plunders = 1 }
	add_to_variable = { FIU.piracy_gains_var = 0.01 }
	if = {
		limit = { NOT = { FIU = { has_dynamic_modifier = { modifier = FIU_piracy_gains_modifier } } } }
		FIU = { add_dynamic_modifier = { modifier = FIU_piracy_gains_modifier } }
	}
	clamp_variable = {
		var = FIU.piracy_gains_var
		min = 0
	}
	hidden_effect = {
		country_event = {
			id = nufiupirate.1000
			days = 1
		}
	}
}

FIU_coup_raise = {
	add_to_variable = { FIU_coup_value = 1 }
	if = {
		limit = {
			check_variable = {
				var = FIU_coup_value

			}

		}

				factor = 1.5

			}
		}
	}
