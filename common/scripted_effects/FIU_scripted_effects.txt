FIU_add_risk_effect = {
	if = {
		limit = { has_country_flag = FIU_coffer_crisis }
		custom_override_tooltip = {
			tooltip = FIU_add_risk_effect_tt
			add_to_variable = { FIU.FIU_risk = temp_FIU_risk }
			clamp_variable = {
				var = FIU.FIU_risk
				min = 0
				max = 1
			}
			FIU_check_for_headsup_gui_effect = yes
		}
	}
}

FIU_check_for_headsup_gui_effect = {
	if = {
		limit = {
			OR = {
				check_variable = { FIU.FIU_risk > 0.8 }
				check_variable = { FIU.FIU_risk = 0.8 }
			}
		}
		set_variable = { FIU.headsup_display_var = 5 }
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { FIU.FIU_risk > 0.6 }
				check_variable = { FIU.FIU_risk = 0.6 }
			}
		}
		set_variable = { FIU.headsup_display_var = 4 }
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { FIU.FIU_risk > 0.4 }
				check_variable = { FIU.FIU_risk = 0.4 }
			}
		}
		set_variable = { FIU.headsup_display_var = 3 }
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { FIU.FIU_risk > 0.2 }
				check_variable = { FIU.FIU_risk = 0.2 }
			}
		}
		set_variable = { FIU.headsup_display_var = 2 }
	}
	else_if = {
		limit = { check_variable = { FIU.FIU_risk < 0.2 } }
		set_variable = { FIU.headsup_display_var = 1 }
	}
}
