FIU_add_risk_effect = {
	custom_override_tooltip = {
		tooltip = FIU_add_risk_effect_tt
		add_to_variable = { FIU.FIU_risk = temp_FIU_risk }
		clamp_variable = {
			var = FIU.FIU_risk
			min = 0
			max = 1
		}
		FIU_check_for_headsup_gui_effect = yes
	}
}

FIU_check_for_headsup_gui_effect = {
	if = {
		limit = {
			OR = {
				check_variable = { FIU.FIU_risk > 0.8 }
				check_variable = { FIU.FIU_risk = 0.8 }
			}
		}
		set_variable = { FIU.headsup_display_var = 5 }
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { FIU.FIU_risk > 0.6 }
				check_variable = { FIU.FIU_risk = 0.6 }
			}
		}
		set_variable = { FIU.headsup_display_var = 4 }
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { FIU.FIU_risk > 0.4 }
				check_variable = { FIU.FIU_risk = 0.4 }
			}
		}
		set_variable = { FIU.headsup_display_var = 3 }
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { FIU.FIU_risk > 0.2 }
				check_variable = { FIU.FIU_risk = 0.2 }
			}
		}
		set_variable = { FIU.headsup_display_var = 2 }
	}
	else_if = {
		limit = { check_variable = { FIU.FIU_risk < 0.2 } }
		set_variable = { FIU.headsup_display_var = 1 }
	}
}

FIU_add_one_successful_plunder = {
	custom_override_tooltip = {
		tooltip = FIU_raid_succeeds
		add_to_variable = { FIU_successful_plunders = 1 }
		add_to_variable = { FIU.piracy_gains_var = 0.01 }
		if = {
			limit = { NOT = { FIU = { has_dynamic_modifier = { modifier = FIU_piracy_gains_modifier } } } }
			FIU = { add_dynamic_modifier = { modifier = FIU_piracy_gains_modifier } }
		}
		set_temp_variable = { temp_clamp_min = FIU.pirate_level }
		divide_temp_variable = { temp_clamp_min = 100 }
		clamp_variable = {
			var = FIU.piracy_gains_var
			min = temp_clamp_min
		}
		country_event = nufiupirate.1000
	}
}

FIU_coup_raise = {
	if = {
		limit = {
			check_variable = { FIU_coup_value < 7 }
			NOT = { has_country_flag = FIU_coups_over }
		}
		custom_override_tooltip = {
			tooltip = FIU_coup_raise_tt
			meta_effect = {
				text = {
					country_event = {
						id = nufiucoup.[COUP_EVENT_NUMBER]
						days = 3
						random_days = 10
						random_hours = 240
					}
				}
				COUP_EVENT_NUMBER = "[?FIU_coup_value]"
			}
		}
	}
	else = {
		add_stability = -0.05
		add_popularity = {
			ideology = national_populist
			popularity = 0.05
		}
	}
	add_to_variable = { FIU_coup_value = 1 }
}

FIU_do_risk_calculation = {
	custom_effect_tooltip = FIU_only_if_failed
	set_temp_variable = { fiu_temp_risk_math = FIU.FIU_RISK }
	add_to_temp_variable = { fiu_temp_risk_math = 1 }
	random_list = {
		25 = {
			custom_override_tooltip = {
				tooltip = FIU_compromise_will_happen_tt
				country_event = {
					id = nufiupirate.0
					days = 7
					random = 200
				}
			}
			modifier = {
				factor = 1.2
				has_country_flag = FIU_readied_raid
			}
			modifier = {
				add = fiu_temp_risk_math
				factor = fiu_temp_risk_math
			}
			modifier = {
				factor = 1.5
				PREV = { FIU_state_is_medium_target = yes }
			}
			modifier = {
				factor = 2
				PREV = { FIU_state_is_large_target = yes }
			}
			modifier = {
				factor = 0.5
				check_variable = { FIU.FIU_RISK < 0.2 }
			}
		}
		75 = {
			modifier = {
				factor = 2.5
				has_country_flag = FIU_readied_raid
			}
			modifier = {
				factor = 1.5
				has_country_flag = FIU_laying_low
			}
			custom_effect_tooltip = FIU_compromise_will_not_happen_tt
		}
	}
	hidden_effect = {
		FIU_check_for_headsup_gui_effect = yes
	}
}
