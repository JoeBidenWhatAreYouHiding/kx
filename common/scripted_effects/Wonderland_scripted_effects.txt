lacey_economy_boost_effect = {
	if = {
		limit = { has_idea = WLD_EcoA }
		swap_ideas = {
			remove_idea = WLD_EcoA
			add_idea = WLD_EcoB
		}
	}
	else_if = {
		limit = { has_idea = WLD_EcoB }
		swap_ideas = {
			remove_idea = WLD_EcoB
			add_idea = WLD_EcoC
		}
	}
	else_if = {
		limit = { has_idea = WLD_EcoC }
		swap_ideas = {
			remove_idea = WLD_EcoC
			add_idea = WLD_EcoD
		}
	}
	else_if = {
		limit = { has_idea = WLD_EcoD }
		swap_ideas = {
			remove_idea = WLD_EcoD
			add_idea = WLD_EcoE
		}
	}
	else_if = {
		limit = { has_idea = WLD_EcoE }
		add_stability = 0.05
	}
}
